2. SUPABASE SQL ŞEMASI (SUPABASE/SCHEMA.SQL BİREBİR REFERANSI)
================================================================================

Tüm Android veri modelleri, DTO'lar ve Retrofit servisleri aşağıdaki PostgreSQL veritabanı şemasına %100 uyumlu olmalıdır:

```sql
-- EXTENSIONS & ENUMS
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

CREATE TYPE public.user_role AS ENUM ('user', 'moderator', 'admin');
CREATE TYPE public.report_status AS ENUM ('pending', 'investigating', 'resolved', 'dismissed');
CREATE TYPE public.reaction_type AS ENUM ('like', 'heart', 'insightful', 'support');

-- 1. PROFILES
CREATE TABLE public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  username TEXT UNIQUE NOT NULL,
  full_name TEXT,
  avatar_url TEXT,
  role public.user_role DEFAULT 'user',
  level INTEGER DEFAULT 1,
  xp INTEGER DEFAULT 0,
  bio TEXT,
  header_url TEXT,
  is_locked BOOLEAN DEFAULT false,
  is_bot BOOLEAN DEFAULT false,
  is_discoverable BOOLEAN DEFAULT true,
  is_verified BOOLEAN DEFAULT false,
  accessibility_prefs JSONB DEFAULT '{"fontSize": 100, "highContrast": false, "dyslexicFont": false, "reducedMotion": false}'::jsonb,
  followers_count INTEGER DEFAULT 0,
  following_count INTEGER DEFAULT 0,
  statuses_count INTEGER DEFAULT 0,
  last_status_at TIMESTAMPTZ,
  last_seen_at TIMESTAMPTZ DEFAULT now(),
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- 2. CATEGORIES
CREATE TABLE public.categories (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name TEXT NOT NULL,
  slug TEXT UNIQUE NOT NULL,
  description TEXT,
  icon TEXT,
  color TEXT,
  parent_id UUID REFERENCES public.categories(id) ON DELETE CASCADE,
  sort_order INTEGER DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- 3. TOPICS
CREATE TABLE public.topics (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  author_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
  category_id UUID REFERENCES public.categories(id) ON DELETE CASCADE NOT NULL,
  title TEXT NOT NULL,
  content TEXT NOT NULL,
  likes_count INTEGER DEFAULT 0,
  comments_count INTEGER DEFAULT 0,
  views_count INTEGER DEFAULT 0,
  is_pinned BOOLEAN DEFAULT false,
  is_locked BOOLEAN DEFAULT false,
  is_solved BOOLEAN DEFAULT false,
  visibility TEXT DEFAULT 'public',
  is_sensitive BOOLEAN DEFAULT false,
  spoiler_text TEXT,
  search_vector TSVECTOR GENERATED ALWAYS AS (to_tsvector('turkish', coalesce(title,'') || ' ' || coalesce(content,''))) STORED,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- 4. COMMENTS
CREATE TABLE public.comments (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  topic_id UUID REFERENCES public.topics(id) ON DELETE CASCADE NOT NULL,
  author_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
  parent_id UUID REFERENCES public.comments(id) ON DELETE CASCADE,
  content TEXT NOT NULL,
  likes_count INTEGER DEFAULT 0,
  is_solution BOOLEAN DEFAULT false,
  visibility TEXT DEFAULT 'public',
  is_sensitive BOOLEAN DEFAULT false,
  spoiler_text TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- 5. REACTIONS
CREATE TABLE public.reactions (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
  topic_id UUID REFERENCES public.topics(id) ON DELETE CASCADE,
  comment_id UUID REFERENCES public.comments(id) ON DELETE CASCADE,
  reaction_type public.reaction_type DEFAULT 'like'::public.reaction_type,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- 6. BOOKMARKS
CREATE TABLE public.bookmarks (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
  topic_id UUID REFERENCES public.topics(id) ON DELETE CASCADE NOT NULL,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- 7. NOTIFICATIONS
CREATE TABLE public.notifications (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
  type TEXT NOT NULL,
  title TEXT NOT NULL,
  body TEXT NOT NULL,
  data JSONB,
  read_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- 8. REPORTS
CREATE TABLE public.reports (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  reporter_id UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  topic_id UUID REFERENCES public.topics(id) ON DELETE CASCADE,
  comment_id UUID REFERENCES public.comments(id) ON DELETE CASCADE,
  reason TEXT NOT NULL,
  status public.report_status DEFAULT 'pending'::public.report_status,
  created_at TIMESTAMPTZ DEFAULT now()
);
```

3. RETROFIT API VE SUPABASE POSTGREST ENTEGRASYONU (KOTLIN)
================================================================================

Retrofit 2 ve OkHttp 4 istemcisi Supabase REST API (PostgREST) ile doğrudan haberleşecek şekilde yapılandırılmalıdır:

1. ZORUNLU HEADER'LAR:
   • `apikey`: `NEXT_PUBLIC_SUPABASE_ANON_KEY`
   • `Authorization`: `Bearer <USER_JWT_TOKEN_OR_ANON_KEY>`
   • `Content-Type`: `application/json`
   • `Prefer`: `return=representation` (INSERT/UPDATE işlemlerinde veri dönüşü için)

2. RETROFIT INTERFACE TANIMLARI (`SupabaseApiService.kt`):

```kotlin
interface SupabaseApiService {
    // --- AUTHENTICATION ---
    @POST("auth/v1/signup")
    suspend fun signUp(@Body body: SignUpRequest): Response<AuthResponse>

    @POST("auth/v1/token?grant_type=password")
    suspend fun signIn(@Body body: SignInRequest): Response<AuthResponse>

    @GET("auth/v1/user")
    suspend fun getCurrentUser(@Header("Authorization") token: String): Response<UserResponse>

    // --- TOPICS (TARTIŞMALAR) ---
    @GET("rest/v1/topics?select=*,author:profiles(*),category:categories(*)&order=created_at.desc")
    suspend fun getTopics(
        @Query("category_id") categoryFilter: String? = null,
        @Query("title") searchFilter: String? = null,
        @Header("Range") range: String = "0-19"
    ): List<TopicDto>

    @GET("rest/v1/topics?select=*,author:profiles(*),category:categories(*)&id=eq.{id}")
    suspend fun getTopicById(@Path("id") id: String): List<TopicDto>

    @POST("rest/v1/topics")
    suspend fun createTopic(@Body topic: CreateTopicRequest): Response<TopicDto>

    // --- COMMENTS (YORUMLAR) ---
    @GET("rest/v1/comments?select=*,author:profiles(*)&topic_id=eq.{topicId}&order=created_at.asc")
    suspend fun getCommentsByTopic(@Path("topicId") topicId: String): List<CommentDto>

    @POST("rest/v1/comments")
    suspend fun createComment(@Body comment: CreateCommentRequest): Response<CommentDto>

    // --- CATEGORIES (KATEGORİLER) ---
    @GET("rest/v1/categories?order=sort_order.asc")
    suspend fun getCategories(): List<CategoryDto>

    // --- PROFILES (PROFILLER) ---
    @GET("rest/v1/profiles?id=eq.{id}")
    suspend fun getProfileById(@Path("id") id: String): List<ProfileDto>

    @PATCH("rest/v1/profiles?id=eq.{id}")
    suspend fun updateProfile(@Path("id") id: String, @Body profile: UpdateProfileRequest): Response<Unit>

    // --- REACTIONS & BOOKMARKS ---
    @POST("rest/v1/reactions")
    suspend fun addReaction(@Body reaction: ReactionRequest): Response<Unit>

    @POST("rest/v1/bookmarks")
    suspend fun addBookmark(@Body bookmark: BookmarkRequest): Response<Unit>
}
```
